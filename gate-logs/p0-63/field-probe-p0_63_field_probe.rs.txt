//! FIELD PROBE (p0-63) — NOT A WORKSPACE PIN. Run at 9c22b84 to seal the field
//! regression at my own bytes, then removed from the tree; the source is kept
//! under `gate-logs/p0-63/` as evidence alongside its log.
//!
//! Reproduces manifold's registry-seat boot shape against a live server over
//! real sockets: enroll (the "attach mailboxes" step), then DRAIN the connection
//! until it reports quiet, then declare ready. The drain read is a pump read —
//! nothing is owed, silence is the expected answer — and after #61 the only door
//! the SDK offers is `RemoteParticipantHandle::receive()`, which now waits out
//! `RESPONSE_DEADLINE` = 60 s. That blows a 30 s boot gate.
//!
//! At 9c22b84 this measures ~60 s and fails the assertion. That is the field
//! shape, at my bytes.

use std::error::Error;
use std::path::Path;
use std::sync::{Arc, Mutex};
use std::time::{Duration, Instant};

use liminal_protocol::wire::{
    ClientRequest, EnrollmentRequest, EnrollmentToken, ParticipantId, ServerValue,
};
use liminal_sdk::{
    ConnectionPoolConfig, ParticipantResumeStore, RemoteConfig, RemoteOperationRecordOutcome,
    RemoteParticipantHandle, RemoteParticipantInbound, RemoteParticipantSendOutcome, SdkError,
};
use liminal_server::config::types::ParticipantConfig;
use liminal_server::config::{LimitsConfig, ServerConfig, ServicesConfig};
use liminal_server::server::connection::{
    ConnectionServices, ConnectionSupervisor, LiminalConnectionServices,
};
use liminal_server::server::listener::ServerListener;

const PROBE_CONVERSATION: u64 = 0x_63_01;

/// The boot budget the field gate enforces, halved: a drain that answers in
/// under this cannot blow a 30 s gate however the seat schedules it.
const BOOT_BUDGET: Duration = Duration::from_secs(10);

/// Frames the drain will read before it gives up calling itself a drain.
const MAX_DRAIN_FRAMES: usize = 64;

const fn participant_config() -> ParticipantConfig {
    ParticipantConfig {
        wire_frame_limit: 65_536,
        attach_receipt_ttl_ms: 60_000,
        receipt_provenance_ttl_ms: 600_000,
        max_live_attach_receipts_server: 1_024,
        max_live_attach_receipts_per_participant: 8,
        max_receipt_provenance_server: 4_096,
        max_receipt_provenance_per_conversation: 256,
        max_receipt_provenance_per_participant: 64,
        max_retired_identity_slots_server: 1_024,
        identity_slots: 4,
        observer_recovery_max_entries: 64,
        max_semantic_conversations_per_connection: 32,
        max_ordinary_record_entries: 1,
        max_ordinary_record_bytes: 131_072,
        max_generated_marker_entries: 1,
        max_generated_marker_bytes: 4_096,
        mandatory_transaction_bound_entries: 4,
        mandatory_transaction_bound_bytes: 16_384,
        full_recovery_claim_entries: 4,
        full_recovery_claim_bytes: 16_384,
        retained_capacity_entries: 2_048,
        retained_capacity_bytes: 16_777_216,
        max_retained_record_rows: 1_024,
        closure_episode_churn_limit: 1_024,
    }
}

#[derive(Debug, Default, Clone)]
struct ProbeStore {
    canonical: Arc<Mutex<Vec<u8>>>,
}

impl ParticipantResumeStore for ProbeStore {
    fn persist(&mut self, canonical_lpcr: &[u8]) -> Result<(), SdkError> {
        let mut guard = self.canonical.lock().map_err(|_| SdkError::Store {
            description: "the probe resume store lock was poisoned".to_owned(),
        })?;
        guard.clear();
        guard.extend_from_slice(canonical_lpcr);
        drop(guard);
        Ok(())
    }
}

fn server_config(store_dir: &Path) -> Result<ServerConfig, Box<dyn Error>> {
    Ok(ServerConfig {
        listen_address: "127.0.0.1:0".parse()?,
        health_listen_address: "127.0.0.1:0".parse()?,
        drain_timeout_ms: 30_000,
        channels: Vec::new(),
        routing_rules: Vec::new(),
        persistence_path: Some(store_dir.to_path_buf()),
        cluster: None,
        auth: None,
        services: ServicesConfig::default(),
        limits: LimitsConfig::default(),
        websocket: None,
        participant: Some(participant_config()),
    })
}

struct ProbeServer {
    listener: Option<ServerListener>,
    supervisor: ConnectionSupervisor,
    address: String,
}

impl ProbeServer {
    fn start(store_dir: &Path) -> Result<Self, Box<dyn Error>> {
        std::fs::create_dir_all(store_dir)?;
        let config = server_config(store_dir)?;
        let services = Arc::new(LiminalConnectionServices::from_config(&config)?);
        let supervisor =
            ConnectionSupervisor::with_services(services as Arc<dyn ConnectionServices>)?;
        let listener = ServerListener::bind(&config, supervisor.clone())?;
        let address = listener.local_addr().to_string();
        Ok(Self {
            listener: Some(listener),
            supervisor,
            address,
        })
    }

    fn dial(&self, channel: &str) -> Result<RemoteConfig, Box<dyn Error>> {
        Ok(RemoteConfig::new(
            self.address.clone(),
            channel,
            PROBE_CONVERSATION.to_string(),
            ConnectionPoolConfig::new(1, 1, 8),
        )?
        .connect_tcp()?)
    }

    fn shutdown(mut self) -> Result<(), Box<dyn Error>> {
        if let Some(listener) = self.listener.take() {
            listener.shutdown()?;
        }
        self.supervisor.shutdown();
        Ok(())
    }
}

fn exchange(
    participant: &RemoteParticipantHandle<ProbeStore>,
    request: ClientRequest,
) -> Result<ServerValue, Box<dyn Error>> {
    let operation = match participant.record_operation(request)? {
        RemoteOperationRecordOutcome::Recorded(operation)
        | RemoteOperationRecordOutcome::Continuous(operation) => operation,
        RemoteOperationRecordOutcome::Refused { request, reason } => {
            return Err(format!("SDK refused outbound request {request:?}: {reason:?}").into());
        }
    };
    match participant.send_operation(operation)? {
        RemoteParticipantSendOutcome::Sent { .. } => {}
        RemoteParticipantSendOutcome::TransportLost { error, .. } => {
            return Err(format!("SDK transport lost while sending: {error}").into());
        }
    }
    for _ in 0..MAX_DRAIN_FRAMES {
        match participant.receive()? {
            RemoteParticipantInbound::Applied { value, .. } => return Ok(value),
            RemoteParticipantInbound::Push { .. } => {}
            refused @ RemoteParticipantInbound::Refused { .. } => {
                return Err(format!("expected an applied server value, got {refused:?}").into());
            }
        }
    }
    Err("no response arrived within the demux bound".into())
}

/// THE FIELD SHAPE. Attach, drain the backlog, declare ready — under a boot
/// budget. The drain is the only step under measurement.
#[test]
fn a_registry_seat_boots_within_its_gate_after_draining_a_quiet_connection()
-> Result<(), Box<dyn Error>> {
    let store_dir = tempfile::tempdir()?;
    let server = ProbeServer::start(&store_dir.path().join("probe-store"))?;
    let config = server.dial("probe-registry")?;
    let participant = RemoteParticipantHandle::new(&config, ProbeStore::default())?;

    // Step 1: attach mailboxes. A real reply-owed exchange; not under
    // measurement, and it must keep its full 60 s protection.
    let enrolled = exchange(
        &participant,
        ClientRequest::Enrollment(EnrollmentRequest {
            conversation_id: PROBE_CONVERSATION,
            enrollment_token: EnrollmentToken::new([0x63; 16]),
        }),
    )?;
    let ServerValue::EnrollBound(bound) = enrolled else {
        return Err(format!("the probe enrollment did not bind: {enrolled:?}").into());
    };
    let _: ParticipantId = bound.participant_id();

    // Step 2: drain the backlog. Nothing is owed; the connection is quiet. This
    // is the pump read, and it is the whole measurement.
    let started = Instant::now();
    let mut drained = 0_usize;
    let mut quiet = false;
    for _ in 0..MAX_DRAIN_FRAMES {
        match participant.receive() {
            Ok(RemoteParticipantInbound::Push { .. }) => drained += 1,
            Ok(other) => {
                return Err(format!("the drain read a correlated frame it never asked for: \
                                    {other:?}")
                .into());
            }
            Err(_) => {
                // The consumer classifies a failed pump read as "quiet" — the
                // classification manifold's drain loop already does.
                quiet = true;
                break;
            }
        }
    }
    let waited = started.elapsed();

    // Step 3: declare ready.
    println!(
        "probe: drained {drained} frames, quiet={quiet}, drain took {:.3}s",
        waited.as_secs_f64()
    );
    assert!(
        quiet,
        "the drain never reached a quiet report after {MAX_DRAIN_FRAMES} frames"
    );
    assert!(
        waited < BOOT_BUDGET,
        "the quiet drain took {:.3}s against a {:.3}s boot budget — a pump read is waiting out \
         the reply-owed response deadline",
        waited.as_secs_f64(),
        BOOT_BUDGET.as_secs_f64()
    );

    drop(participant);
    drop(config);
    server.shutdown()?;
    Ok(())
}
